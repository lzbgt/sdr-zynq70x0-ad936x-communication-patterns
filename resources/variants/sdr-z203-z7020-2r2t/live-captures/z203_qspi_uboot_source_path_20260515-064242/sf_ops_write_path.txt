   499			byte_addr = offset % page_size;
   500			chunk_len = min(len - actual, (size_t)(page_size - byte_addr));
   501
   502			if (flash->spi->max_write_size)
   503				chunk_len = min(chunk_len,
   504						(size_t)flash->spi->max_write_size);

   507				spi_flash_addr(write_addr, cmd, 1);

   510				spi_flash_addr(write_addr, cmd, 0);
   511				cmdlen = SPI_FLASH_CMD_LEN;
